"""Reprepare pinned public tools; preserve every process in EDA-004G only."""
import os
import runpy
import sys
from pathlib import Path
from capture import run, save, digest, utc

ROOT = Path(__file__).resolve().parent
BASE = Path('/tmp/eda004g')
PRIOR = ROOT.parents[2] / 'tscircuit/qualification/eda-004d'

if __name__ == '__main__':
    BASE.mkdir(exist_ok=False)
    os.environ['EDA004D_WORK_DIR'] = str(BASE)
    os.environ['EDA004D_EVIDENCE_DIR'] = str(ROOT / 'evidence')
    sys.path.insert(0, str(PRIOR))
    runpy.run_path(str(PRIOR / 'install-kicad.py'), run_name='__main__')
    archive = ROOT.parent / 'eda-004f/evidence/library-source.tar.gz'
    assert digest(archive) == 'fa7508829945457f9824ae3490bddfb81718f3cacedaeaa6ee6515c030ef398c'
    source = BASE / 'library-source'
    source.mkdir()
    run('library-extract', ['tar', '-xzf', str(archive), '-C', str(source)], BASE)
    run('python-venv', ['python3', '-m', 'venv', str(BASE / 'venv')], BASE)
    pip = str(BASE / 'venv/bin/pip')
    run('dependencies-install', [pip, 'install', '--require-hashes', '-r', str(ROOT.parent / 'eda-004f/requirements-lock.txt'), '--report', str(ROOT / 'evidence/pip-install-report.json')], BASE, 240)
    run('library-install', [pip, 'install', '--no-deps', str(source), '--report', str(ROOT / 'evidence/library-install-report.json')], BASE, 120)
    run('dependencies', [pip, 'freeze', '--all'], BASE)
    url = 'https://ppa.launchpadcontent.net/kicad/kicad-9.0-releases/ubuntu/pool/main/k/kicad-symbols/kicad-symbols_9.0.9~ubuntu24.04.1_all.deb'
    package = BASE / 'kicad-symbols.deb'
    run('symbol-package-download', ['curl', '-fL', url, '-o', str(package)], BASE, 180)
    assert digest(package) == '73bc5b5f7ada14abcbe324c5f289c6cf807db379a9ffaaaecc877df1b619bd0a'
    run('symbol-package-extract', ['dpkg-deb', '-x', str(package), str(BASE / 'symbols')], BASE)
    save(ROOT / 'evidence/preparation.json', dict(finished_utc=utc(), library_commit='505bca23e91cf3fa6e3b66976e529e775eada3ba', library_source_archive_sha256=digest(archive), symbols_url=url, symbols_sha256=digest(package), same_environment_confirmation=False, full_environment_confirmation=False))
